%% COL 3 — OUTER LOOP  (x = 6.8)
\node[draw, rectangle, text width=3.6cm, align=left, inner sep=4pt,
      anchor=north west] (C) at (6.8cm,-0.6cm)
{
  \textbf{OUTER LOOP}\\[3pt]
  \textbf{Step 1 — Pre-sort by length:}\\
  \texttt{sorted(queries, key=len)}\\
  R3 hints first (most targeted),\\
  then R4 combos, R2 RQs, R1 topic\\[3pt]
  \textbf{Step 2 — Iterate:}\\
  \texttt{for query in sorted\_queries:}\\
  \quad $\to$ enter inner loop\\[3pt]
  \textbf{Step 3 — Saturation check}\\
  \quad (after each query)\\
  yield\_rate = n\_new / n\_raw\\
  rolling window = last 3 queries\\
  if avg\_yield $<$ 5\%:\\
  \quad \textbf{exit loop early}\\
  \quad (remaining queries skipped)\\[3pt]
  \textit{pipeline.py \_search\_all()}
};

%% COL 4 — INNER LOOP  (x = 10.8)
\node[draw, rectangle, text width=4.4cm, align=left, inner sep=4pt,
      anchor=north west] (D) at (10.8cm,-0.6cm)
{
  \textbf{INNER LOOP + SOURCE SEARCH}\\[3pt]
  \texttt{for src in [arXiv, S2, OpenAlex]:}\\[3pt]
  \textbf{Failure gate:}\\
  if consecutive\_failures $\geq$ 3:\\
  \quad skip this source (all rem. queries)\\
  success resets counter to 0\\[3pt]
  \textbf{arXiv} (Atom XML, max 25)\\
  -- title/abstract keyword search\\[2pt]
  \textbf{Semantic Scholar} (REST, max 25)\\
  -- fields: title, authors, year,\\
  \quad abstract, externalIds, openAccess\\[2pt]
  \textbf{OpenAlex} (REST, max 25)\\
  -- filter: from\_year / to\_year\\
  -- fields: title, authorships, abstract\\[3pt]
  error $\to$ failures++, log, continue\\[3pt]
  \textit{arxiv.py / semantic\_scholar.py}\\
  \textit{openalexapi.py}
};

%% COL 5 — INCR. DEDUPE  (x = 15.6)
\node[draw, rectangle, text width=3.0cm, align=left, inner sep=4pt,

{

  Runs \textit{inside} inner loop\\
  (per source, per query)\\[3pt]
  \textbf{Priority order:}\\
  1. DOI\\
  2. arXiv ID\\
  3. OpenAlex ID\\
  4. Semantic Scholar ID\\
  5. title $|$ year $|$ first-author\\[3pt]
  Duplicate: merge sources,\\
  identifiers, abstract, URLs\\[3pt]
  Returns \texttt{n\_new}\\
  $\to$ used by outer loop\\
  \quad for yield rate\\[3pt]
  \textit{dedupe.py}\\
  \textit{IncrementalDeduplicator}
};
